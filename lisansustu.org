# Simple example where we have two hypotheses
import numpy as np

# Prior probabilities for fair (H1) and biased (H2) coin
P_H1 = 0.5  # Prior for H1: Coin is fair
P_H2 = 0.5  # Prior for H2: Coin is biased (P(Heads) = 0.7)

# Likelihood of the data (coin flip result) under each hypothesis
def likelihood(data, hypothesis):
    if hypothesis == "H1":  # Fair coin
        return 0.5
    elif hypothesis == "H2":  # Biased coin
        return 0.7 if data == 'H' else 0.3

# Update beliefs using Bayes' Theorem
def bayesian_update(prior_H1, prior_H2, data):
    # Likelihoods
    likelihood_H1 = likelihood(data, "H1")
    likelihood_H2 = likelihood(data, "H2")

    # Total evidence P(D)
    P_data = likelihood_H1 * prior_H1 + likelihood_H2 * prior_H2

    # Posterior for H1 (fair coin)
    posterior_H1 = (likelihood_H1 * prior_H1) / P_data

    # Posterior for H2 (biased coin)
    posterior_H2 = (likelihood_H2 * prior_H2) / P_data

    return posterior_H1, posterior_H2

# Simulate a series of coin flips
# coin_flips = ['H', 'T', 'H', 'H', 'T'] # Example sequence of flips
# Generate 100 random coin flips (1 for heads, 0 for tails)
coin_flips = np.random.choice(['T', 'H'], size=20)

P_H1_posterior = P_H1
P_H2_posterior = P_H2

# Perform Bayesian updates after each flip
for flip in coin_flips:
    P_H1_posterior, P_H2_posterior = bayesian_update(P_H1_posterior, P_H2_posterior, flip)
    print(f"After observing {flip}: P(H1 | Data) = {P_H1_posterior:.4f}, P(H2 | Data) = {P_H2_posterior:.4f}")
#+END_SRC

#+BEGIN_SRC python :results output
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import poisson
# Data
ClicksPerTenSeconds = np.arange(18)  # Number of counts per 10-second interval
ObservedCounts = [1, 6, 11, 28, 56, 105, 126, 146, 164, 161, 123, 101, 74, 53, 23, 15, 9, 5]

plt.style.use('dark_background')
# Calculate the sample mean (initial estimate of lambda)
sample_mean = np.average(ClicksPerTenSeconds, weights=ObservedCounts)
# Number of bootstrap samples
num_bootstrap = 1000
bootstrap_lambdas = []
# Bootstrap procedure
for _ in range(num_bootstrap):
    # Generate a synthetic dataset using the estimated lambda
    synthetic_data = np.random.poisson(lam=sample_mean, size=np.sum(ObservedCounts))

    # Bin the synthetic data to match the observed format (number of counts per 10-second interval)
    synthetic_hist, _ = np.histogram(synthetic_data, bins=np.arange(19))

    # Calculate the new lambda estimate from the synthetic data
    new_lambda = np.average(ClicksPerTenSeconds, weights=synthetic_hist)

    # Store the new lambda estimate
    bootstrap_lambdas.append(new_lambda)
# Plot the distribution of bootstrap lambda estimates
plt.figure(figsize=(10, 6))
plt.hist(bootstrap_lambdas, bins=30, color='green', edgecolor='black', alpha=0.7)
plt.title('Bootstrap Distribution of Lambda Estimates', fontsize=14)
plt.xlabel('Lambda Estimate', fontsize=12)
plt.ylabel('Frequency', fontsize=12)
plt.grid(True)
# Calculate the sample mean
sample_mean = np.average(ClicksPerTenSeconds, weights=ObservedCounts)
# Fit a Poisson distribution using the sample mean as lambda (rate parameter)
poisson_dist = poisson.pmf(ClicksPerTenSeconds, mu=sample_mean) * np.sum(ObservedCounts)
# Plotting the observed data
plt.figure(figsize=(10, 6))
plt.plot(ClicksPerTenSeconds, ObservedCounts, marker='o', linestyle='-', color='blue', label='Observed Counts')
# Plot the Poisson distribution
plt.plot(ClicksPerTenSeconds, poisson_dist, marker='x', linestyle='--', color='red', label='Poisson Fit')
# Adding labels and title
plt.title('Observed Alpha Particle Emissions per 10-Second Interval', fontsize=14)
plt.xlabel('Number of Clicks per 10-Second Interval', fontsize=12)
plt.ylabel('Observed Count', fontsize=12)
plt.grid(True)

# Maximum Likelihood Estimation (MLE)
def negative_log_likelihood(theta):
    a, b = theta
    y_pred = a * x + b
    residuals = y - y_pred
    return np.sum(residuals**2) / (2 * sigma_noise**2)

  import random
  import math
  from collections import Counter


  # -------------------------
  # Shared utility functions
  # -------------------------

  def squared_euclidean_distance(a, b):
      """Return the squared Euclidean distance between two vectors."""
      if len(a) != len(b):
          raise ValueError("Vectors must have the same number of dimensions.")

      return sum((x - y) ** 2 for x, y in zip(a, b))


  def euclidean_distance(a, b):
      return math.sqrt(squared_euclidean_distance(a, b))


  # -------------------------
  # K-Means Clustering
  # -------------------------

  class KMeans:
      def __init__(self, n_clusters, max_iterations=100, tolerance=1e-4,
                   random_state=None):
          if n_clusters <= 0:
              raise ValueError("n_clusters must be greater than zero.")

          self.n_clusters = n_clusters
          self.max_iterations = max_iterations
          self.tolerance = tolerance
          self.random_state = random_state

          self.centroids = []
          self.labels = []

      def _validate_data(self, X):
          if not X:
              raise ValueError("X cannot be empty.")

          dimensions = len(X[0])

          if dimensions == 0:
              raise ValueError("Data points must have at least one dimension.")

          for point in X:
              if len(point) != dimensions:
                  raise ValueError("All points must have the same number of dimensions.")

      def _initialize_centroids(self, X):
          if self.n_clusters > len(X):
              raise ValueError("n_clusters cannot be greater than the number of samples.")

          rng = random.Random(self.random_state)

          # Choose distinct random points as initial centroids
          indices = rng.sample(range(len(X)), self.n_clusters)
          return [list(X[i]) for i in indices]

      def _assign_clusters(self, X, centroids):
          labels = []

          for point in X:
              distances = [
                  squared_euclidean_distance(point, centroid)
                  for centroid in centroids
              ]

              nearest_cluster = distances.index(min(distances))
              labels.append(nearest_cluster)

          return labels

      def _recalculate_centroids(self, X, labels, old_centroids):
          dimensions = len(X[0])
          new_centroids = []

          for cluster_id in range(self.n_clusters):
              cluster_points = [
                  X[i] for i in range(len(X))
                  if labels[i] == cluster_id
              ]

              if not cluster_points:
                  # Keep the previous centroid if the cluster is empty
                  new_centroids.append(old_centroids[cluster_id])
                  continue

              centroid = [
                  sum(point[dimension] for point in cluster_points)
                  / len(cluster_points)
                  for dimension in range(dimensions)
              ]

              new_centroids.append(centroid)

          return new_centroids

      def fit(self, X):
          self._validate_data(X)

          centroids = self._initialize_centroids(X)

          for _ in range(self.max_iterations):
              labels = self._assign_clusters(X, centroids)
              new_centroids = self._recalculate_centroids(
                  X, labels, centroids
              )

              centroid_shift = max(
                  euclidean_distance(old, new)
                  for old, new in zip(centroids, new_centroids)
              )

              centroids = new_centroids

              if centroid_shift <= self.tolerance:
                  break

          self.centroids = centroids
          self.labels = self._assign_clusters(X, self.centroids)

          return self

      def predict(self, X):
          if not self.centroids:
              raise RuntimeError("Call fit() before predict().")

          return self._assign_clusters(X, self.centroids)

      def fit_predict(self, X):
          self.fit(X)
          return self.labels


  # -------------------------
  # K-Nearest Neighbors
  # -------------------------

  class KNN:
      def __init__(self, k=3):
          if k <= 0:
              raise ValueError("k must be greater than zero.")

          self.k = k
          self.X_train = []
          self.y_train = []

      def fit(self, X, y):
          if not X:
              raise ValueError("X cannot be empty.")

          if len(X) != len(y):
              raise ValueError("X and y must contain the same number of samples.")

          dimensions = len(X[0])

          for point in X:
              if len(point) != dimensions:
                  raise ValueError("All points must have the same number of dimensions.")

          if self.k > len(X):
              raise ValueError("k cannot be greater than the number of training samples.")

          self.X_train = [list(point) for point in X]
          self.y_train = list(y)

          return self

      def _nearest_neighbors(self, point):
          distances = []

          for train_point, label in zip(self.X_train, self.y_train):
              distance = squared_euclidean_distance(point, train_point)
              distances.append((distance, label))

          distances.sort(key=lambda item: item[0])

          return distances[:self.k]

      def _vote(self, neighbors):
          labels = [label for _, label in neighbors]
          counts = Counter(labels)

          highest_count = max(counts.values())

          # Resolve ties by choosing the label of the closest tied neighbor
          tied_labels = {
              label for label, count in counts.items()
              if count == highest_count
          }

          for _, label in neighbors:
              if label in tied_labels:
                  return label

      def predict_one(self, point):
          if not self.X_train:
              raise RuntimeError("Call fit() before predict().")

          neighbors = self._nearest_neighbors(point)
          return self._vote(neighbors)

      def predict(self, X):
          return [self.predict_one(point) for point in X]
